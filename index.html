/* * ===================================================== * EXIF EXTRACTION LÓGICA (ATUALIZADA) * ===================================================== */ function extractExif(img) { EXIF.getData(img, function() { // Extração das tags EXIF padrão let model = EXIF.getTag(this, "Model"); let focal = EXIF.getTag(this, "FocalLength"); let fnum = EXIF.getTag(this, "FNumber"); let iso = EXIF.getTag(this, "ISOSpeedRatings") || EXIF.getTag(this, "ISO"); let exp = EXIF.getTag(this, "ExposureTime"); let exifData = []; // 1. Camera Model if (model) { exifData.push(model.trim()); } // 2. Focal Length (mm) if (focal) { exifData.push(Number(focal) + " mm"); } // 3. Aperture if (fnum) { exifData.push("f/" + Number(fnum).toFixed(1)); } // 4. ISO if (iso) { exifData.push("ISO " + iso); } // 5. Shutter Speed if (exp) { let e = Number(exp); if (e >= 1) { exifData.push(e + " s"); } else { exifData.push("1/" + Math.round(1 / e) + " s"); } } // Junta todas as informações encontradas separadas por " | " let exifString = exifData.join(" | "); // Texto de fallback em inglês if (!exifString) { exifString = "No EXIF data available"; } // Salva no atributo HTML para o Modal (Lightbox) usar depois img.setAttribute("data-exif", exifString); // Atualiza a legenda na grade de fotos let container = img.closest('.gallery-item'); if (container) { let exifSpan = container.querySelector('.caption-exif'); if (exifSpan) { exifSpan.innerText = exifString; } } }); }
